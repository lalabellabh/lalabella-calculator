// ==================== Lalabella Item Inventory (Ribbon, Vase, Fixed Assets, etc.) ==================== // Sheet expected columns in "ItemInventory": Item Code | Item Name | Item Type | Category | // Qty | Unit | Price (BHD) | Min Stock | Photo Link (Drive) | Date Added | Notes // A second sheet "ItemReleaseHistory" logs every release/use (mirrors Chocolate's Release DataBase). const ITEM_PHOTOS_FOLDER_ID = '1DBOB2sUjrVTFavxBFrbsaSycHKnzzgXv'; // "LalabellaItemPhotos" function itemCol(headers, name) { const target = String(name).trim().toLowerCase(); for (let i = 0; i < headers.length; i++) { if (String(headers[i]).trim().toLowerCase() === target) return i; } return -1; } function getItemNextCounter(sheet, keyName) { const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === keyName) { const val = Number(data[i][1]) || 1; sheet.getRange(i + 1, 2).setValue(val + 1); return val; } } const nextRow = data.length + 1; sheet.getRange(nextRow, 1).setValue(keyName); sheet.getRange(nextRow, 2).setValue(2); return 1; } function generateItemProductCode(itemName, settingsSheet) { const prefix = String(itemName || 'ITM').trim().toUpperCase().replace(/[^A-Z]/g, '').slice(0, 3) || 'ITM'; const num = getItemNextCounter(settingsSheet, 'Next ItemInventory Code'); return prefix + '-' + String(num).padStart(5, '0'); } // Saves a base64-encoded image (already compressed client-side) into // the dedicated LalabellaItemPhotos Drive folder, and returns a // direct-view link — the Sheet only ever stores this link, never the // image data itself, so the sheet stays light no matter how many // items get a photo. function saveItemPhotoToDrive(base64Data, itemCode) { if (!base64Data) return ''; const folder = DriveApp.getFolderById(ITEM_PHOTOS_FOLDER_ID); // Strip a data: URL prefix if the client sent one (e.g. "data:image/jpeg;base64,...."). const commaIdx = base64Data.indexOf(','); const cleanBase64 = commaIdx !== -1 ? base64Data.slice(commaIdx + 1) : base64Data; const bytes = Utilities.base64Decode(cleanBase64); const blob = Utilities.newBlob(bytes, 'image/jpeg', itemCode + '.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // Every Drive hotlink URL format (uc?id=, thumbnail?id=, // lh3.googleusercontent.com/d/) turned out to render fine when // opened directly but still fail when embedded as on this // page — a known, inconsistent Drive hotlinking quirk. Storing just // the bare file ID sidesteps the whole problem: the actual image // bytes get fetched server-side (via DriveApp, which always has // access) and sent back as inline data, never as an external link // the browser has to fetch on its own. return file.getId(); } // Turns a stored Photo Link value into a data: URI with the actual // image bytes inline, wherever the browser can just use it as `src`. // Handles three shapes for backward compatibility with items saved // before this fix: a bare file ID (current format), a full Drive URL // containing "id=FILE_ID" (thumbnail/uc formats), or a // lh3.googleusercontent.com/d/FILE_ID link. function itemPhotoToDataUri(photoLink) { if (!photoLink) return ''; let fileId = String(photoLink).trim(); const idMatch = fileId.match(/[?&]id=([a-zA-Z0-9_-]+)/) || fileId.match(/\/d\/([a-zA-Z0-9_-]+)/); if (idMatch) fileId = idMatch[1]; try { const file = DriveApp.getFileById(fileId); const blob = file.getBlob(); return 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes()); } catch (err) { return ''; // File missing/inaccessible — show no image rather than erroring the whole list. } } function doPost(e) { return doGet(e); } function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); // Staff name suggestions for the "Entered By / Received By / Released // By" dropdowns — pulled from whoever has actually been entered on // past Item Inventory activity, so the list grows on its own as // people use the system (no separate admin screen needed just to // seed it). if (action === 'getItemStaff') { const names = new Set(); const sheet = ss.getSheetByName('ItemInventory'); if (sheet) { const data = sheet.getDataRange().getValues(); const headers = data[0]; const iEnteredBy = itemCol(headers, 'Entered By'); if (iEnteredBy !== -1) { for (let i = 1; i < data.length; i++) { const v = String(data[i][iEnteredBy] || '').trim(); if (v) names.add(v); } } } const histSheet = ss.getSheetByName('ItemReleaseHistory'); if (histSheet) { const data = histSheet.getDataRange().getValues(); const headers = data[0]; const iBy = itemCol(headers, 'Released By'); if (iBy !== -1) { for (let i = 1; i < data.length; i++) { const v = String(data[i][iBy] || '').trim(); if (v) names.add(v); } } } return ContentService.createTextOutput(JSON.stringify([...names].sort())) .setMimeType(ContentService.MimeType.JSON); } if (action === 'getItemInventory') { const sheet = ss.getSheetByName('ItemInventory'); const data = sheet.getDataRange().getValues(); const headers = data.shift(); const photoCol = 'Photo Link (Drive)'; const items = data .filter(row => row[0]) .map(row => { let obj = {}; headers.forEach((h, i) => obj[String(h).trim()] = row[i]); if (obj[photoCol]) obj[photoCol] = itemPhotoToDataUri(obj[photoCol]); return obj; }); return ContentService.createTextOutput(JSON.stringify(items)) .setMimeType(ContentService.MimeType.JSON); } // Overview for the Item Dashboard — totals, Consumable vs Fixed // Asset split, low-stock list, and recent release activity, mirroring // the same shape of dashboard the Chocolate/Flower systems already have. if (action === 'getItemDashboard') { const sheet = ss.getSheetByName('ItemInventory'); const data = sheet.getDataRange().getValues(); const headers = data.shift(); const iName = itemCol(headers, 'Item Name'); const iCode = itemCol(headers, 'Item Code'); const iType = itemCol(headers, 'Item Type'); const iQty = itemCol(headers, 'Qty'); const iUnit = itemCol(headers, 'Unit'); const iPrice = itemCol(headers, 'Price (BHD)'); const iMinStock = itemCol(headers, 'Min Stock'); const iPhoto = itemCol(headers, 'Photo Link (Drive)'); const items = data.filter(row => row[iName]); let totalItems = 0, consumableCount = 0, fixedAssetCount = 0, totalInventoryValue = 0; const lowStockList = []; const allItemsList = []; items.forEach(row => { totalItems++; const type = String(row[iType] || 'Consumable').trim(); if (type === 'Fixed Asset') fixedAssetCount++; else consumableCount++; const qty = Number(row[iQty]) || 0; const price = Number(row[iPrice]) || 0; totalInventoryValue += qty * price; const minStock = row[iMinStock]; const hasMinStock = minStock !== '' && minStock !== undefined && minStock !== null; if (hasMinStock && qty < Number(minStock)) { lowStockList.push({ name: row[iName], code: row[iCode], qty, unit: row[iUnit], minStock: Number(minStock) }); } allItemsList.push({ name: row[iName], code: row[iCode], type, qty, unit: row[iUnit], price, photo: row[iPhoto] ? itemPhotoToDataUri(row[iPhoto]) : '' }); }); lowStockList.sort((a, b) => (a.qty - a.minStock) - (b.qty - b.minStock)); allItemsList.sort((a, b) => String(a.name).localeCompare(String(b.name))); // Recent release history (last 15), same pattern as Chocolate's // Release DataBase / Release history views. let recentReleases = []; const histSheet = ss.getSheetByName('ItemReleaseHistory'); if (histSheet) { const histData = histSheet.getDataRange().getValues(); const histHeaders = histData.shift(); recentReleases = histData .filter(row => row[0]) .map(row => { let obj = {}; histHeaders.forEach((h, i) => obj[String(h).trim()] = row[i]); return obj; }) .reverse() .slice(0, 15); } return ContentService.createTextOutput(JSON.stringify({ totalItems, consumableCount, fixedAssetCount, totalInventoryValue, lowStockList, allItemsList, recentReleases })).setMimeType(ContentService.MimeType.JSON); } // Adds a brand-new item to the catalog (the "baseline" — first time // this item has ever been entered, so whatever Qty is given here IS // the starting quantity). Auto-generates a Product Code if the // caller doesn't supply one, and saves + links a photo if provided. // Fixes/corrects an EXISTING item's details — the counterpart to // addNewItem, for when a Category, Price, Photo, or Product Code // needs updating after the fact (same pattern as Chocolate's own // "Fix an Existing Catalog Item" tool). if (action === 'updateItemInventory') { const sheet = ss.getSheetByName('ItemInventory'); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const data = sheet.getDataRange().getValues(); const iName = itemCol(headers, 'Item Name'); const itemName = (e.parameter.itemName || '').trim(); if (!itemName) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item name is required.' })) .setMimeType(ContentService.MimeType.JSON); } let row = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][iName]).trim().toLowerCase() === itemName.toLowerCase()) { row = i; break; } } if (row === -1) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item not found: ' + itemName })) .setMimeType(ContentService.MimeType.JSON); } const sheetRow = row + 1; const setCol = (colName, value) => { const idx = itemCol(headers, colName); if (idx !== -1) sheet.getRange(sheetRow, idx + 1).setValue(value); }; if (e.parameter.itemType) setCol('Item Type', e.parameter.itemType); if (e.parameter.category) setCol('Category', e.parameter.category); if (e.parameter.unit) setCol('Unit', e.parameter.unit); if (e.parameter.price !== undefined && e.parameter.price !== '') setCol('Price (BHD)', Number(e.parameter.price) || 0); if (e.parameter.minStock !== undefined) setCol('Min Stock', e.parameter.minStock === '' ? '' : Number(e.parameter.minStock)); if (e.parameter.itemCode) setCol('Item Code', e.parameter.itemCode); if (e.parameter.photoBase64) { const itemCode = e.parameter.itemCode || data[row][itemCol(headers, 'Item Code')]; setCol('Photo Link (Drive)', saveItemPhotoToDrive(e.parameter.photoBase64, itemCode)); } return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } if (action === 'addNewItem') { const sheet = ss.getSheetByName('ItemInventory'); const settingsSheet = ss.getSheetByName('Settings') || ss.insertSheet('Settings'); const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0]; const itemName = (e.parameter.itemName || '').trim(); if (!itemName) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item name is required' })) .setMimeType(ContentService.MimeType.JSON); } // Guard against accidentally re-adding an item that already exists — // point the caller to Receiving instead, which is the correct way // to add MORE quantity to an existing item. const data = sheet.getDataRange().getValues(); const iName = itemCol(headers, 'Item Name'); for (let i = 1; i < data.length; i++) { if (String(data[i][iName]).trim().toLowerCase() === itemName.toLowerCase()) { return ContentService.createTextOutput(JSON.stringify({ error: 'This item already exists (' + data[i][itemCol(headers, 'Item Code')] + '). Use Receiving to add more quantity instead.' })).setMimeType(ContentService.MimeType.JSON); } } let itemCode = (e.parameter.itemCode || '').trim(); if (!itemCode) { itemCode = generateItemProductCode(itemName, settingsSheet); } let photoLink = ''; if (e.parameter.photoBase64) { photoLink = saveItemPhotoToDrive(e.parameter.photoBase64, itemCode); } const row = []; const setIf = (colName, value) => { const idx = itemCol(headers, colName); if (idx !== -1) row[idx] = value; }; setIf('Item Code', itemCode); setIf('Item Name', itemName); setIf('Item Type', e.parameter.itemType || 'Consumable'); // Consumable | Fixed Asset setIf('Category', e.parameter.category || ''); setIf('Qty', Number(e.parameter.qty) || 0); // this first Qty IS the baseline setIf('Unit', e.parameter.unit || 'Piece'); setIf('Price (BHD)', Number(e.parameter.price) || 0); setIf('Min Stock', e.parameter.minStock !== undefined && e.parameter.minStock !== '' ? Number(e.parameter.minStock) : ''); setIf('Photo Link (Drive)', photoLink); setIf('Date Added', new Date()); setIf('Notes', e.parameter.notes || ''); setIf('Entered By', e.parameter.enteredBy || ''); sheet.appendRow(headers.map((h, i) => row[i] !== undefined ? row[i] : '')); return ContentService.createTextOutput(JSON.stringify({ success: true, itemCode: itemCode, photoLink: photoLink })) .setMimeType(ContentService.MimeType.JSON); } // Receiving: adds MORE quantity to an item that already exists. // (A brand-new item goes through addNewItem instead, above.) if (action === 'receiveItem') { const sheet = ss.getSheetByName('ItemInventory'); const data = sheet.getDataRange().getValues(); const headers = data[0]; const iName = itemCol(headers, 'Item Name'); const iQty = itemCol(headers, 'Qty'); const itemName = (e.parameter.itemName || '').trim(); const qtyReceived = Number(e.parameter.qty) || 0; if (!itemName || qtyReceived <= 0) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item name and a Qty greater than 0 are required.' })) .setMimeType(ContentService.MimeType.JSON); } let row = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][iName]).trim().toLowerCase() === itemName.toLowerCase()) { row = i; break; } } if (row === -1) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item not found. Use "Add New Item" first if this is a brand-new item.' })).setMimeType(ContentService.MimeType.JSON); } const currentQty = Number(data[row][iQty]) || 0; const newQty = currentQty + qtyReceived; sheet.getRange(row + 1, iQty + 1).setValue(newQty); // Log this receiving event with its own date and who received it — // the main ItemInventory row only ever holds the CURRENT total, so // without this log there'd be no record of when/by whom each // addition happened (same gap Chocolate/Flower avoid with their // own Receiving history sheets). let recvHistSheet = ss.getSheetByName('ItemReceivingHistory'); if (!recvHistSheet) { recvHistSheet = ss.insertSheet('ItemReceivingHistory'); recvHistSheet.appendRow(['Date', 'Item Code', 'Item Name', 'Qty Received', 'Received By']); } recvHistSheet.appendRow([ e.parameter.date || new Date(), data[row][itemCol(headers, 'Item Code')], itemName, qtyReceived, e.parameter.receivedBy || '' ]); return ContentService.createTextOutput(JSON.stringify({ success: true, itemName: itemName, previousQty: currentQty, received: qtyReceived, newQty: newQty })).setMimeType(ContentService.MimeType.JSON); } // Release: subtracts quantity when an item is sold/used for an // event, and logs the release the same way Chocolate's Release // DataBase does — a running history, not just a live number. if (action === 'releaseItem') { const sheet = ss.getSheetByName('ItemInventory'); let histSheet = ss.getSheetByName('ItemReleaseHistory'); if (!histSheet) { histSheet = ss.insertSheet('ItemReleaseHistory'); histSheet.appendRow(['Date', 'Item Code', 'Item Name', 'Qty Released', 'Reason/Event', 'Released By']); } const data = sheet.getDataRange().getValues(); const headers = data[0]; const iName = itemCol(headers, 'Item Name'); const iCode = itemCol(headers, 'Item Code'); const iQty = itemCol(headers, 'Qty'); const itemName = (e.parameter.itemName || '').trim(); const qtyReleased = Number(e.parameter.qty) || 0; if (!itemName || qtyReleased <= 0) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item name and a Qty greater than 0 are required.' })) .setMimeType(ContentService.MimeType.JSON); } let row = -1; for (let i = 1; i < data.length; i++) { if (String(data[i][iName]).trim().toLowerCase() === itemName.toLowerCase()) { row = i; break; } } if (row === -1) { return ContentService.createTextOutput(JSON.stringify({ error: 'Item not found.' })) .setMimeType(ContentService.MimeType.JSON); } const currentQty = Number(data[row][iQty]) || 0; if (qtyReleased > currentQty) { return ContentService.createTextOutput(JSON.stringify({ error: 'Not enough stock. Available: ' + currentQty + ', tried to release: ' + qtyReleased })).setMimeType(ContentService.MimeType.JSON); } const newQty = currentQty - qtyReleased; sheet.getRange(row + 1, iQty + 1).setValue(newQty); histSheet.appendRow([ e.parameter.date || new Date(), data[row][iCode], itemName, qtyReleased, e.parameter.reason || '', e.parameter.releasedBy || '' ]); // ---- Odoo live-deduction hook (inactive until IT provides API access) ---- // Once Odoo credentials are available, this is where a call to // Odoo's JSON-RPC/REST inventory endpoint would go, deducting the // same qtyReleased there too — right after the local Sheet update // succeeds, so the two systems never drift apart. Left as a no-op // for now so this feature works standalone in the meantime. // syncReleaseToOdoo(data[row][iCode], qtyReleased); return ContentService.createTextOutput(JSON.stringify({ success: true, itemName: itemName, previousQty: currentQty, released: qtyReleased, newQty: newQty })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({ error: 'Unknown action' })) .setMimeType(ContentService.MimeType.JSON); }
